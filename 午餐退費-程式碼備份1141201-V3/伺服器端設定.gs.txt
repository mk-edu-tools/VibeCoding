// --- 伺服器端設定 ---

// 處理所有 POST 請求
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000); // 防止多人同時寫入衝突

  try {
    var request = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    // --- 1. 讀取不供餐日設定 (直接讀取 Sheet 資料) ---
    // 前端一載入就會呼叫此功能，取得目前的假日清單
    if (request.action === 'getHolidays') {
      var settingSheet = getOrCreateSheet(ss, '系統參數_假日設定'); // 這裡對應您 Sheet 的分頁名稱
      var holidays = [];
      var lastRow = settingSheet.getLastRow();
      
      if (lastRow > 0) {
        // 讀取 A 欄所有資料
        var data = settingSheet.getRange(1, 1, lastRow, 1).getValues();
        // 格式化日期並過濾空值
        holidays = data.map(function(row) { 
          var val = row[0];
          if (val instanceof Date) {
            // 轉換為 YYYY/M/D 格式
            return val.getFullYear() + '/' + (val.getMonth() + 1) + '/' + val.getDate();
          }
          // 如果是純文字，也嘗試回傳，前端會自行判斷
          return val;
        }).filter(function(d) { return d; });
      }
      
      return responseJSON({ result: "success", holidays: holidays });
    }

    // --- 2. 一般退費申請 (寫入申請表) ---
    var sheet = getOrCreateSheet(ss, '退費申請');
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(['申請時間', '停餐事由', '班級座號/職稱', '姓名', '申請人葷素', '申請人數', '停餐日期', '預計停餐天數', '預計退費金額', '參加社團']);
    }

    var timestamp = new Date();
    var count = parseInt(request.studentCount) || 1;
    var days = parseInt(request.totalDays) || 0;
    var amount = days * count * 60;

    sheet.appendRow([
      timestamp,
      request.reason,
      request.classSeat,
      request.name,
      request.diet,
      count,
      request.dateInput,
      days,
      amount,
      request.clubs
    ]);

    // --- 寄信通知 (請自行修改 Email) ---
    var adminEmail = "請在此輸入呂小姐的Email"; 
    if (adminEmail && adminEmail.indexOf("@") > -1 && adminEmail !== "請在此輸入呂小姐的Email") {
      try {
        MailApp.sendEmail({
          to: adminEmail,
          subject: "【午餐退費】" + request.classSeat + " " + request.name,
          body: "收到新的申請。\n日期：" + request.dateInput + "\n金額：" + amount
        });
      } catch(e) {}
    }

    return responseJSON({ result: "success" });

  } catch (error) {
    return responseJSON({ result: "error", error: error.toString() });
  } finally {
    lock.releaseLock();
  }
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateSheet(ss, sheetName) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) { sheet = ss.insertSheet(sheetName); }
  return sheet;
}

function doGet(e) {
  return HtmlService.createHtmlOutput("API Service Running");
}

    <script>
      // ==========================================
      // 模擬 Google Apps Script 環境 (預覽模式用)
      // ==========================================
      if (typeof google === 'undefined') {
        console.warn('正在預覽模式運行，使用模擬資料。');
        var mockRunner = {
          withSuccessHandler: function(func) { this.successHandler = func; return this; },
          withFailureHandler: function(func) { this.errorHandler = func; return this; },
          getInitData: function() {
             setTimeout(() => {
               this.successHandler({
                  studentMap: { 
                    '101': { '1': '王小明 (模擬)', '2': '李小華 (模擬)', '3': '陳大文 (模擬)' }, 
                    '102': { '1': '張三 (模擬)' } 
                  },
                  teacherMap: { '101': '陳老師', '102': '林老師' }
               });
             }, 800);
          },
          handleBorrow: function(data) {
             setTimeout(() => this.successHandler({success:true}), 600);
          },
          getStudentActiveRecords: function(c, s) {
             // 模擬一些未歸還的資料
             setTimeout(() => {
               const mockRecords = (s == '1') ? [
                 {id:'mock-1', time:'12/01 12:00', classId:c, seatNo:s, studentName:'王小明 (模擬)', teacherName:'陳老師', items:'碗, 筷子'}
               ] : [];
               this.successHandler(mockRecords);
             }, 600);
          },
          handleReturn: function(id) {
             setTimeout(() => this.successHandler({success:true}), 600);
          }
        };
        window.google = { script: { run: mockRunner } };
      }
      // ==========================================

      let globalData = {};
      let selectedItems = new Set();
      let currentStudent = { classId: '', seatNo: '', name: '', teacher: '' };

      window.onload = function() {
        // 呼叫後端 (或模擬後端)
        google.script.run
          .withSuccessHandler(initForm)
          .withFailureHandler(showError)
          .getInitData();
      };

      function initForm(data) {
        globalData = data;
        const classSelect = document.getElementById('classSelect');
        const classes = Object.keys(data.studentMap).sort();
        classes.forEach(cls => {
          let option = document.createElement('option');
          option.value = cls;
          option.text = cls + ' 班';
          classSelect.appendChild(option);
        });
        document.getElementById('loading').classList.add('hidden');
        document.getElementById('mainForm').classList.remove('hidden');
      }

      function updateStudentList() {
        const classId = document.getElementById('classSelect').value;
        const seatSelect = document.getElementById('seatSelect');
        seatSelect.innerHTML = '<option value="">請選擇</option>';
        resetForm();

        if (classId && globalData.studentMap[classId]) {
          seatSelect.disabled = false;
          // 自然排序座號
          const seats = Object.keys(globalData.studentMap[classId]).sort((a,b) => parseInt(a) - parseInt(b));
          seats.forEach(seat => {
            let option = document.createElement('option');
            option.value = seat;
            option.text = seat + ' 號';
            seatSelect.appendChild(option);
          });
        } else {
          seatSelect.disabled = true;
        }
      }

      function identifyStudent() {
        const classId = document.getElementById('classSelect').value;
        const seatNo = document.getElementById('seatSelect').value;

        if (classId && seatNo) {
          const studentName = globalData.studentMap[classId][seatNo];
          const teacherName = globalData.teacherMap[classId] || '未設定';

          currentStudent = { classId, seatNo, name: studentName, teacher: teacherName };

          document.getElementById('displayStudentName').innerText = studentName;
          document.getElementById('displayTeacherName').innerText = teacherName;
          document.getElementById('infoArea').classList.remove('hidden');
          
          checkButtonState();
          fetchStudentRecords();
        } else {
          document.getElementById('infoArea').classList.add('hidden');
          currentStudent = { classId: '', seatNo: '', name: '', teacher: '' };
        }
      }

      function toggleItem(element, item) {
        if (selectedItems.has(item)) {
          selectedItems.delete(item);
          element.classList.remove('selected');
        } else {
          selectedItems.add(item);
          element.classList.add('selected');
        }
        checkButtonState();
      }

      function checkButtonState() {
        const btn = document.getElementById('submitBtn');
        const hasStudent = currentStudent.name !== '';
        const hasItems = selectedItems.size > 0;

        if (hasStudent) {
          if (hasItems) {
            btn.disabled = false;
            btn.classList.remove('bg-gray-400');
            btn.classList.add('bg-blue-600', 'hover:bg-blue-700', 'hover:shadow-xl');
            btn.innerText = '確認借用';
          } else {
            btn.disabled = true;
            btn.classList.add('bg-gray-400');
            btn.classList.remove('bg-blue-600', 'hover:bg-blue-700', 'hover:shadow-xl');
            btn.innerText = '請選擇餐具';
          }
        } else {
          btn.disabled = true;
          btn.innerText = '請先選擇班級座號';
        }
      }

      function resetForm() {
        selectedItems.clear();
        document.querySelectorAll('.item-btn').forEach(el => el.classList.remove('selected'));
        document.getElementById('infoArea').classList.add('hidden');
        document.getElementById('returnSection').classList.add('hidden');
        currentStudent = { classId: '', seatNo: '', name: '', teacher: '' };
        checkButtonState();
      }

      function submitBorrow() {
        if (selectedItems.size === 0) return;
        const btn = document.getElementById('submitBtn');
        const originalText = btn.innerText;
        btn.disabled = true;
        btn.innerText = '處理中...';

        const formData = {
          classId: currentStudent.classId,
          seatNo: currentStudent.seatNo,
          studentName: currentStudent.name,
          teacherName: currentStudent.teacher,
          items: Array.from(selectedItems)
        };

        google.script.run
          .withSuccessHandler(function(response) {
            Swal.fire({
              icon: 'success',
              title: '借用成功',
              text: '記得要還喔！',
              timer: 2000,
              showConfirmButton: false
            });
            selectedItems.clear();
            document.querySelectorAll('.item-btn').forEach(el => el.classList.remove('selected'));
            checkButtonState();
            fetchStudentRecords();
          })
          .withFailureHandler(function(err) {
            showError(err);
            btn.disabled = false;
            btn.innerText = originalText;
          })
          .handleBorrow(formData);
      }

      function fetchStudentRecords() {
        if (!currentStudent.classId) return;
        google.script.run
          .withSuccessHandler(renderRecords)
          .getStudentActiveRecords(currentStudent.classId, currentStudent.seatNo);
      }

      function renderRecords(records) {
        const list = document.getElementById('recordsList');
        const section = document.getElementById('returnSection');
        list.innerHTML = '';

        if (records.length > 0) {
          section.classList.remove('hidden');
          records.forEach(rec => {
            const div = document.createElement('div');
            div.className = 'flex justify-between items-center bg-white p-4 rounded-xl border-l-4 border-l-orange-500 border-gray-200 shadow-sm';
            
            div.innerHTML = `
              <div class="flex-1">
                <div class="text-sm text-gray-500 mb-2 flex flex-wrap gap-2 items-center">
                   <span class="bg-gray-100 px-3 py-1 rounded-full text-gray-600"><i class="far fa-clock mr-1"></i>${rec.time}</span>
                   <span class="bg-blue-100 text-blue-800 px-3 py-1 rounded-full font-bold">${rec.classId}班 ${rec.seatNo}號</span>
                </div>
                <div class="text-base mb-1">
                    <span class="font-bold text-gray-800 text-xl mr-2">${rec.studentName}</span>
                    <span class="text-gray-500">(${rec.teacherName}老師)</span>
                </div>
                <div class="font-bold text-orange-600 text-lg flex items-center mt-2">
                   <i class="fa-solid fa-basket-shopping mr-2"></i>${rec.items}
                </div>
              </div>
              <button onclick="returnItem('${rec.id}', this)" class="ml-4 bg-green-500 text-white px-6 py-4 rounded-xl text-lg font-bold hover:bg-green-600 transition-colors shadow-md whitespace-nowrap active:scale-95 transform">
                歸還
              </button>
            `;
            list.appendChild(div);
          });
        } else {
          section.classList.add('hidden');
        }
      }

      function returnItem(recordId, btnElement) {
        btnElement.innerText = '...';
        btnElement.disabled = true;
        google.script.run
          .withSuccessHandler(function() {
             Swal.fire({
              icon: 'success',
              title: '歸還完成',
              toast: true,
              position: 'top-end',
              showConfirmButton: false,
              timer: 1500
            });
            fetchStudentRecords();
          })
          .withFailureHandler(showError)
          .handleReturn(recordId);
      }

      function showError(err) {
        console.error(err);
        Swal.fire({
          icon: 'error',
          title: '發生錯誤',
          text: err.message || '系統發生未知錯誤'
        });
      }
    </script>
